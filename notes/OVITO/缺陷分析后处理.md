```shell
export LIBGL_ALWAYS_SOFTWARE=1
export QT_QPA_PLATFORM="offscreen"

CONTAINER_IMAGE="/opt/software/ovito-3.12.3/bin/ovito_python.sif"
HOST_HOME="$HOME"
CONTAINER_BIND_ROOT="/app"

original_dir="$(pwd -P)"
container_dir="/app${original_dir#$HOME}"
target_dir="$(realpath "../gif")" # GIF 文件存放路径
mkdir -p "$target_dir"
container_target_dir="/app${target_dir#$HOME}"

get_parent_dir() {
  local path="$1"
  local level="$2"
  for ((i=0; i<level; i++)); do
    path="$(dirname "$path")"
  done
  basename "$path"
}

# 默认轨迹文件存放路径格式：./pot/ene/idx/radiation/cascade/dump.xyz
find "$original_dir" -name 'dump.xyz' -print0 | while IFS= read -r -d '' file; do
  idx=$(get_parent_dir "$file" 3)
  ene=$(get_parent_dir "$file" 4)
  pot=$(get_parent_dir "$file" 5)
  container_input="/app${file#$HOME}"
  container_output="${container_target_dir}/${pot}_${ene}_${idx}.gif"
  if apptainer exec --bind ~/:/app "$CONTAINER_IMAGE" python $container_dir/cascade.py --input_file $container_input --output_file $container_output; then
    echo "成功生成: ${target_dir}/${pot}_${ene}_${idx}.gif"
  else
    echo "错误: 文件处理失败 - $file"
  fi
done
```

其中，`cascade.py` 脚本如下所示：

```python
import argparse
from ovito.io import import_file
from ovito.vis import Viewport, TachyonRenderer
from ovito.modifiers import WignerSeitzAnalysisModifier, ExpressionSelectionModifier, DeleteSelectedModifier, AssignColorModifier

def GIF_generator(input_file, output_file):
    """
    Wigner-Seitz 缺陷分析后处理函数
    """
    # 加载轨迹文件
    pipeline = import_file(input_file, multiple_frames=True)

    # 使用 Wigner-Seitz 分析缺陷
    wigner_seitz = WignerSeitzAnalysisModifier(
        output_displaced=False
    )
    pipeline.modifiers.append(wigner_seitz)

    # 删除 Occupancy == 1 的原子
    pipeline.modifiers.append(ExpressionSelectionModifier(expression="Occupancy == 1"))
    pipeline.modifiers.append(DeleteSelectedModifier())

    # 为 Occupancy > 1 的原子（间隙原子）着色为红色
    pipeline.modifiers.append(ExpressionSelectionModifier(expression="Occupancy > 1"))
    pipeline.modifiers.append(AssignColorModifier(color=(1.0, 0.0, 0.0)))

    # 为 Occupancy == 0 的原子（空位）着色为蓝色
    pipeline.modifiers.append(ExpressionSelectionModifier(expression="Occupancy == 0"))
    pipeline.modifiers.append(AssignColorModifier(color=(0.0, 0.0, 1.0)))

    # 将 pipeline 添加到场景中
    pipeline.add_to_scene()

    # 设置正视图
    vp = Viewport(type=Viewport.Type.Front)
    vp.zoom_all()

    # 设置渲染器
    renderer = TachyonRenderer()

    # 生成 GIF 动画
    try:
        vp.render_anim(
            filename=output_file,
            size=(800, 800),  # 分辨率
            renderer=renderer,
            fps=5,  # 帧率
            range=(0, pipeline.source.num_frames - 1)  # 帧范围
        )
        print(f"保存成功！")
    except Exception as e:
        print(f"保存 GIF 时发生错误：{e}")
    finally:
        pipeline.remove_from_scene()
        del pipeline

def main():
    parser = argparse.ArgumentParser(description="根据输入的轨迹文件生成 GIF 动图")
    parser.add_argument("--input_file", type=str, default="dump.xyz", help="轨迹文件名")
    parser.add_argument("--output_file", type=str, default="dump.gif", help="生成的 GIF 文件名")

    args = parser.parse_args()
    GIF_generator(args.input_file, args.output_file)


if __name__ == "__main__":
    main()
```
